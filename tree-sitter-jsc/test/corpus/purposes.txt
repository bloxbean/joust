package t
contract C(stakeSelf) {
  params { gov: Credential, previous: Hash }
  state { admin: Key, successor: Bytes }
  withdraw collect(amount: Int) authorized params.gov { send(wallet(admin), Value.lovelace(amount)) }
  certify cert(c: TxCert) authorized params.gov = true
  vote vote(voter: Voter, action: GovernanceActionId, vote: Vote) authorized params.gov { action.govActionIx == 0 }
  propose rail(p: ProposalProcedure) = p.deposit > 0
  init migrate from params.previous(old: Old.State) { return { admin: old.admin, successor: # } }
  destroy upgrade() only admin migrates to successor { }
}
