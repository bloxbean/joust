package p
enum Phase { Open, Closed { day: Int, by: Key } }
contract C {
  state { p: Phase, o: Option<Int> }
  init() { return { p: Open, o: None } }
  destroy d() only n {
    match p {
      Closed { by } => { pay(by, balance) }
      _ => pay(caller, balance)
    }
  }
  view v(): Int = match o { Some(x) => x, None => 0 }
  view w(): Int = match p { Closed { day: d } => d; Open => 0 }
}

jsc 2
package t
contract C {
  state { n: Int }
  init() { return { n: 0 } }
  action a(v: Int) payable open {
    send(wallet(caller), Value.lovelace(v))
    mint(#01, v)
    burn(#02, 1)
    accept(funds.received)
    match guard {
      KeyCred { key } => send(wallet(key), Value.none())
      ScriptCred { hash } => mint(hash, 1)
    }
    return state
  }
}
