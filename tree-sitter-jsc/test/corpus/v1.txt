package org.example.state

contract Jar(maxDatumBytes = 192) {
    state owner: Key
    state points: Map<Key, Int>
    init() payable { owner = caller }
    action tip(to: Key) payable batched {
        require received >= 1 ada "tip at least 1 ADA"
        points[to] += 1
        if has(points, to) { this.owner = owner } else { delete points[to] }
    }
    destroy close() only owner { pay balance to owner }
    view reached(): Bool = points[owner] >= 2
}
