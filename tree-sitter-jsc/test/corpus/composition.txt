jsc 2
package t
contract C {
  state { owner: Key }
  action give(to: Address) open {
    deposit(params.pot, Value.lovelace(1), caller)
    send(to, Value.lovelace(2), Note { by: caller })
    send(to, Value.lovelace(2))
    match funds.deposits.head {
      Some(d) => deposit(params.pot, d.value, d.memo),
      None => emit(Nothing)
    }
    return state
  }
}
